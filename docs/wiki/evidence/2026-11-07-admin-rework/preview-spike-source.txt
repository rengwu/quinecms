<script>
const $ = (id) => document.getElementById(id);
const work = $('work'), pv = $('pv'), vp = $('vp'), site = $('site'), bar = $('bar');
const devices = { desktop: [1280, 800], mobile: [390, 844] };
const BAR = 32, MIN = 200;
const saved = (window.fungiPreview && window.fungiPreview.state) || {};
const s = Object.assign({ mode: 'split', device: 'desktop', split: 460, nav: true, x: null, y: null, fw: { desktop: 380, mobile: 210 } }, saved);
const save = () => window.fungiPreview && window.fungiPreview.setState(JSON.parse(JSON.stringify(s)));
function float() {
  const [W, H] = devices[s.device], ww = work.clientWidth, wh = work.clientHeight;
  let w = Math.max(MIN, Math.min(s.fw[s.device], ww - 16, (wh - 16 - BAR) * W / H));
  const h = w * H / W + BAR;
  if (s.x === null) { s.x = ww - w - 16; s.y = wh - h - 16; }
  s.x = Math.max(0, Math.min(s.x, ww - w)); s.y = Math.max(0, Math.min(s.y, wh - h));
  s.fw[s.device] = w;
  return { w, h };
}
function render() {
  const [W, H] = devices[s.device];
  document.getElementById('app').classList.toggle('nav-off', !s.nav);
  work.dataset.mode = s.mode;
  for (const b of $('modes').children) b.classList.toggle('on', b.dataset.mode === s.mode);
  for (const b of $('devices').children) b.classList.toggle('on', b.dataset.device === s.device);
  if (s.mode === 'off') return;
  let scale, vh;
  if (s.mode === 'split') {
    s.split = Math.max(280, Math.min(s.split, work.clientWidth - 320));
    work.style.setProperty('--pw', s.split + 'px');
    pv.style.cssText = '';
    const sw = $('stage').clientWidth - 16, sh = $('stage').clientHeight - 16;
    scale = Math.min(1, sw / W); vh = sh;
  } else {
    const { w, h } = float();
    pv.style.cssText = `left:${s.x}px;top:${s.y}px;width:${w}px;height:${h}px`;
    scale = w / W; vh = H * scale;
  }
  vp.style.width = W * scale + 'px'; vp.style.height = vh + 'px';
  site.style.width = W + 'px'; site.style.height = vh / scale + 'px';
  site.style.transform = `scale(${scale})`;
  $('scale').textContent = Math.round(scale * 100) + '%';
}
function drag(el, start) {
  el.addEventListener('pointerdown', (e) => {
    if (e.button !== 0 || e.target.closest('button')) return;
    e.preventDefault();
    el.setPointerCapture(e.pointerId);
    const move = start(e);
    work.classList.add('busy');
    const up = () => { work.classList.remove('busy'); el.removeEventListener('pointermove', onMove); el.removeEventListener('pointerup', up); el.removeEventListener('pointercancel', up); save(); };
    const onMove = (m) => { move(m.clientX - e.clientX, m.clientY - e.clientY); render(); };
    el.addEventListener('pointermove', onMove); el.addEventListener('pointerup', up); el.addEventListener('pointercancel', up);
  });
}
drag($('divider'), () => { const start = s.split; return (dx) => { s.split = start - dx; }; });
drag(bar, () => { if (s.mode !== 'float') return () => {}; const x = s.x, y = s.y; return (dx, dy) => { s.x = x + dx; s.y = y + dy; }; });
for (const handle of document.querySelectorAll('.h')) drag(handle, () => {
  const c = handle.dataset.c, [W, H] = devices[s.device];
  const x = s.x, y = s.y, w = s.fw[s.device], h = w * H / W + BAR;
  return (dx, dy) => {
    const gx = c.includes('e') ? dx : -dx, gy = c.includes('s') ? dy : -dy;
    let nw = w + (gx + gy * W / H) / 2;
    const maxW = c.includes('e') ? work.clientWidth - x : x + w;
    const maxH = c.includes('s') ? work.clientHeight - y : y + h;
    nw = Math.max(MIN, Math.min(nw, maxW, (maxH - BAR) * W / H));
    const nh = nw * H / W + BAR;
    s.fw[s.device] = nw;
    s.x = c.includes('w') ? x + w - nw : x;
    s.y = c.includes('n') ? y + h - nh : y;
  };
});
$('modes').addEventListener('click', (e) => { const b = e.target.closest('button'); if (b) { s.mode = b.dataset.mode; render(); save(); } });
$('devices').addEventListener('click', (e) => { const b = e.target.closest('button'); if (b) { s.device = b.dataset.device; render(); save(); } });
$('close').addEventListener('click', () => { s.mode = 'off'; render(); save(); });
$('navtoggle').addEventListener('click', () => { s.nav = !s.nav; render(); save(); });
$('heading').addEventListener('input', (e) => { $('sh').textContent = e.target.value; });
$('eyebrow').addEventListener('input', (e) => { $('se').textContent = e.target.value; });
new ResizeObserver(render).observe(work);
render();
</script>
